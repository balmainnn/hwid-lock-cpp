cmake_minimum_required(VERSION 3.20)
project(hwid_lock_demo VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This demo targets Windows because it reads the Windows MachineGuid.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hwid_lock STATIC src/hwid_lock.cpp)
target_include_directories(hwid_lock PUBLIC include)
target_link_libraries(hwid_lock PRIVATE advapi32 bcrypt)
target_compile_definitions(hwid_lock PRIVATE UNICODE _UNICODE)
add_executable(hwid-demo src/main.cpp)
target_link_libraries(hwid-demo PRIVATE hwid_lock)
